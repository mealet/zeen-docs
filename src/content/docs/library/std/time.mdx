> **Usage Example**
> ```zeen
> use std.time;
>
> fn main() {
>   let start = Instant.now();
>
>   Time.sleep(Duration.from_millis(20));
>
>   @println("elapsed: {}", start.elapsed());
>   @println("now: {}", Time.now());
> }
> ```
