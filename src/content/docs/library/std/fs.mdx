> **Usage Example**
> ```zeen
> use std.fs;
> 
> fn main() {
>   // create dir
>   Fs.create_dir("/tmp/demo").unwrap();
>   // write file
>   Fs.write("/tmp/demo/hello.txt", "hello fs").unwrap();
> 
>   // read file
>   let text = Fs.read_to_string("/tmp/demo/hello.txt").unwrap();
>   @println("{}", text);
> 
>   // list dir
>   let entries = Fs.read_dir("/tmp/demo").unwrap();
>   @println("entries: {}", entries.len());
> }
> ```
