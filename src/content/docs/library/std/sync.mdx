> **Usage Example**
> ```zeen
> use std.sync;
>
> fn main() {
>   let counter = Arc.new(Mutex.new(0));
>
>   {
>     let guard = counter.get_mut().lock();
>     *guard += 1;
>   };
>
>   @println("count: {}", *counter.get_mut().lock());
> }
> ```
