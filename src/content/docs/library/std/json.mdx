> **Usage Example**
> ```zeen
> use std.json;
>
> fn main() {
>   let value = JsonValue.parse(r#"{"name":"zeen","tags":[1,2]}"#).unwrap();
>
>   let name = value.get("name").unwrap();
>   @println("name: {}", name.as_str().unwrap());
>
>   @println("tags: {}", value.get("tags").unwrap().len());
>   @println("{}", value.stringify());
> }
> ```
