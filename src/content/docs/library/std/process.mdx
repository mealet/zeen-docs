> **Usage Example**
> ```zeen
> use std.process;
>
> fn main() {
>   let out = Command.new("echo").arg("hello").output().unwrap();
>
>   @println("status: {}", out.status.success());
>   @println("stdout: {}", out.stdout.trim_end());
> }
> ```
>
> ```zeen
> use std.process;
>
> fn main() {
>   for (arg : Process.args_iter()) {
>     @println("{}", arg);
>   }
> }
> ```
