> **Usage Example**
> ```zeen
> use std.net;
> 
> fn main() {
>   // bind listener
>   let addr = SocketAddr.loopback(0);
>   let listener = TcpListener.bind(&addr).unwrap();
>   let local = listener.local_addr().unwrap();
> 
>   // connect client and accept server side
>   let client = TcpStream.connect(&local).unwrap();
>   let server = listener.accept().unwrap();
> 
>   // send and read
>   client.write_all("ping").unwrap();
>
>   let buf = String.new();
>   server.read_exact(&buf, 4).unwrap();
>
>   @println("got: {}", buf);
> }
> ```
